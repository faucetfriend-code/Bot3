Pythonimport argparse
import pandas as pd
from strategies import STRATEGY_REGISTRY, optimize_strategy, compute_metrics
from data_manager import get_candles

parser = argparse.ArgumentParser()
parser.add_argument("--ticker", default="BTC-USDT")
parser.add_argument("--start", default="2018-01-01")
parser.add_argument("--end", default="2026-03-15")
parser.add_argument("--sd_min", type=float, default=0.5)
parser.add_argument("--sd_max", type=float, default=6.0)
parser.add_argument("--sd_step", type=float, default=0.1)
parser.add_argument("--full_grid", action="store_true")
parser.add_argument("--max_combinations", type=int, default=50000)
parser.add_argument("--output", default="vwap_full_sweep_2018_now.csv")
args = parser.parse_args()

# Load full data once
df = get_candles(args.ticker, "5m", args.start, args.end)

results = []
sd_values = [round(x, 1) for x in list(range(int(args.sd_min*10), int(args.sd_max*10)+1))/10]

for sd in sd_values:
    print(f"Running full sweep for SD = {sd}")
    # Force sd_threshold and run full grid for all other variables
    _, grid, defaults = STRATEGY_REGISTRY["VWAP Scalping"]
    grid["sd_threshold"] = [sd]  # lock this SD
    
    best = optimize_strategy(df, cutoff=0.08, strategy_name="VWAP Scalping")
    eq, trd = STRATEGY_REGISTRY["VWAP Scalping"][0](df, 0.08, **best)
    m = compute_metrics(eq, trd, bars_per_year=105120)
    
    results.append({
        "SD": sd,
        **best,
        "Trades": m["n_trades"],
        "WinRate": m["win_rate_pct"],
        "PF": m["profit_factor"],
        "Net%": m["total_return_pct"],
        "Sharpe": m["sharpe"]
    })

pd.DataFrame(results).to_csv(args.output, index=False)
print(f"✅ Full VWAP sweep complete → {args.output}")
