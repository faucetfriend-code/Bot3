def run_vwap_scalping(
    df: pd.DataFrame,
    cutoff: float,
    sd_threshold: float = 2.5,
    atr_stop: float = 0.7,
    atr_target: float = 3.8,
    ema_fast: int = 9,
    ema_slow: int = 20,
    use_volume_filter: bool = True,
    volume_mult: float = 2.0,
    use_trailing_stop: bool = True,
    trailing_atr: float = 1.2,
    adx_max: float = 20.0,
    rsi_max: float = 40.0,
    entry_mode: str = "bull_pullback",   # ← NEW DEFAULT
    pullback_bars: int = 3,              # NEW: how many 1h bars to look back for higher high
    df_exit: pd.DataFrame | None = None,
) -> tuple[pd.Series, list]:
    """
    VWAP Scalping — BULL PULLBACK MODE (catches dips in uptrends)
    Uses already-gathered 1h data to confirm "pullback after higher high".
    """
    fc = apply_butterworth(df["Close"], cutoff)
    vwap, vs = compute_vwap_rolling(df["High"], df["Low"], fc, df["Volume"], 20)
    atr = compute_atr(df["High"], df["Low"], df["Close"], 14)
    adx = compute_adx(df["High"], df["Low"], df["Close"], 14)
    rsi = compute_rsi(fc, 14)

    # Higher-TF data (already in your backtester)
    df_1h = df.resample('1H').agg({'Open':'first','High':'max','Low':'min','Close':'last','Volume':'sum'}).dropna()
    ema9_1h = compute_ema(df_1h["Close"], 9)
    ema21_1h = compute_ema(df_1h["Close"], 21)
    vwap_1h = compute_vwap_rolling(df_1h["High"], df_1h["Low"], df_1h["Close"], df_1h["Volume"], 20)[0]

    # Forward-fill to 5m
    vwap_1h_ff = vwap_1h.reindex(df.index, method='ffill')
    ema9_1h_ff = ema9_1h.reindex(df.index, method='ffill')
    ema21_1h_ff = ema21_1h.reindex(df.index, method='ffill')

    # Main 5m loop
    prices = df["Close"].values.astype(float)
    highs = df["High"].values.astype(float)
    lows = df["Low"].values.astype(float)
    opens = df["Open"].values.astype(float)
    volumes = df["Volume"].values.astype(float)
    n = len(prices)

    equity_arr = np.ones(n, dtype=float)
    curr_equity = 1.0
    in_pos = [False]
    side = [""]
    entry_eq = [1.0]
    sl = [0.0]
    tp_ref = [0.0]
    closed_trades: list[float] = []

    bar_dur = df.index[1] - df.index[0] if len(df) > 1 else pd.Timedelta(minutes=5)

    for i in range(1, n):
        vw = vwap.iloc[i]; vsd = vs.iloc[i]
        at = atr.iloc[i]
        adx_v = adx.iloc[i]
        r = rsi.iloc[i]
        p = prices[i]
        o = opens[i]
        v = volumes[i]
        va = df["Volume"].rolling(20).mean().iloc[i]

        if any(np.isnan(x) for x in (vw, vsd, at, adx_v, r)):
            equity_arr[i] = curr_equity
            continue

        ranging = adx_v < adx_max
        volume_surge = v >= volume_mult * va if use_volume_filter else True
        rsi_long_ok = r < rsi_max
        lower_band = vw - sd_threshold * vsd

        # === NEW BULL PULLBACK LOGIC ===
        is_bull_trend = ema9_1h_ff.iloc[i] > ema21_1h_ff.iloc[i]
        recent_higher_high = False
        for k in range(1, pullback_bars + 1):
            if i - k * 12 >= 0:  # 12 bars ≈ 1h
                if highs[i] > highs[i - k * 12]:
                    recent_higher_high = True
                    break

        if in_pos[0]:
            # existing exit code unchanged
            curr_equity *= p / prices[i-1]
            nxt = df.index[i] + bar_dur
            curr_equity = _check_exit_multires(curr_equity, entry_eq[0], highs[i], lows[i],
                sl[0], tp_ref[0], side[0], closed_trades, in_pos,
                bar_ts=df.index[i], next_bar_ts=nxt, df_exit=df_exit)
            # trailing stop unchanged
        else:
            long_signal = False
            if entry_mode == "bull_pullback" and is_bull_trend and recent_higher_high:
                if p < lower_band and ranging and rsi_long_ok and volume_surge and p < vwap_1h_ff.iloc[i]:
                    long_signal = True

            if long_signal:
                entry_p = o
                sl_p = entry_p - atr_stop * at
                tp_p = entry_p + atr_target * at
                curr_equity = _open_long(curr_equity, entry_eq, sl, tp_ref, entry_p, sl_p, tp_p, in_pos, side)

        equity_arr[i] = curr_equity

    # final close logic unchanged
    return pd.Series(equity_arr, index=df.index), closed_trades
