```javascript
// Enhanced loadAccountInfo function
async function loadAccountInfo() {
    const accountDisplay = document.getElementById('envAccountPublicKey');
    if (!accountDisplay) {
        console.warn('envAccountPublicKey element not found');
        return;
    }

    // Set initial loading state
    accountDisplay.textContent = 'Loading...';
    accountDisplay.style.color = '';

    try {
        const response = await safeApiCall(`${API_BASE}/config`);
        if (response.success && response.data && response.data.account_public_key) {
            const accountKey = response.data.account_public_key;
            accountDisplay.textContent = accountKey.slice(0, 8) + '...' + accountKey.slice(-8);
            accountDisplay.title = accountKey;
            accountDisplay.style.color = '#28a745'; // Green for success
            console.log('Account info loaded successfully');
        } else {
            accountDisplay.textContent = 'Not configured';
            accountDisplay.style.color = '#dc3545';
        }
    } catch (error) {
        console.warn('Failed to load account info:', error);
        accountDisplay.textContent = 'Error loading';
        accountDisplay.style.color = '#dc3545';
    }
}
```
